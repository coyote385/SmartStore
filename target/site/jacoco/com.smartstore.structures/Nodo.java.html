package com.smartstore.structures;

/**
 * Representa un nodo de una lista enlazada.
 *
 * @author Jonathan Mendez
 * @version 1.0
 */
public class Nodo<T> {

    private T dato;
    private Nodo<T> siguiente;

    /**
     * Constructor del nodo.
     *
     * @param dato Dato almacenado.
     */
    public Nodo(T dato) {
        this.dato = dato;
        this.siguiente = null;
    }

    public T getDato() {
        return dato;
    }

    public void setDato(T dato) {
        this.dato = dato;
    }

    public Nodo<T> getSiguiente() {
        return siguiente;
    }

    public void setSiguiente(Nodo<T> siguiente) {
        this.siguiente = siguiente;
    }
}
